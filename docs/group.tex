\section{Khó khăn và giải pháp}

\subsection{Thread Synchronization}
\begin{itemize}
    \item \textbf{Vấn đề}: Race conditions khi nhiều threads truy cập vector clock và buffer đồng thời
    \item \textbf{Giải pháp}: Sử dụng \texttt{threading.Lock} để đảm bảo thread-safe operations
    \item \textbf{Kết quả}: Không có race conditions hoặc data corruption trong quá trình chạy
\end{itemize}

\subsection{Message Ordering}
\begin{itemize}
    \item \textbf{Vấn đề}: Messages đến không theo thứ tự do network delay ngẫu nhiên
    \item \textbf{Giải pháp}: Implement buffer với mechanism check điều kiện delivery liên tục
    \item \textbf{Kết quả}: 100\% messages được deliver theo đúng thứ tự nhân quả
\end{itemize}

\subsection{Debugging với Multiple Processes}
\begin{itemize}
    \item \textbf{Vấn đề}: Khó debug và trace issues với 15 processes chạy đồng thời
    \item \textbf{Giải pháp}: Implement chi tiết logging system với timestamps và structured format
    \item \textbf{Kết quả}: Dễ dàng trace và debug issues thông qua log files
\end{itemize}
