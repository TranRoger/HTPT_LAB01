\section{Tổng quan đồ án}

\subsection{Mục tiêu}
Đồ án này nhằm cài đặt và minh họa thuật toán \textbf{SES (Schiper-Eggli-Sandoz)} để đảm bảo thứ tự nhân quả (causal ordering) trong hệ thống phân tán. Chương trình bao gồm 15 processes, mỗi process tự động gửi 150 messages đến mỗi process còn lại với thời gian phát sinh ngẫu nhiên.

\subsection{Yêu cầu đề bài}
\begin{enumerate}
    \item 15 processes chạy đồng thời, có thể trên 1 hoặc nhiều máy tính
    \item Mỗi process gửi 150 messages đến mỗi process khác
    \item Thời gian phát sinh messages ngẫu nhiên (configurable)
    \item Hiển thị rõ ràng việc buffering và delivery của messages
    \item Log file chi tiết cho mỗi process
    \item Chỉ ra khi message được buffer/delivery và lý do
    \item Hiển thị cập nhật vector clock
\end{enumerate}

\section{Cơ sở lý thuyết}

\subsection{Thuật toán SES}

Thuật toán SES (Schiper-Eggli-Sandoz) là một thuật toán đảm bảo thứ tự nhân quả trong hệ thống phân tán bằng cách sử dụng \textbf{Vector Clock}.

\subsubsection{Vector Clock}
Vector Clock là một cơ chế theo dõi quan hệ nhân quả giữa các sự kiện trong hệ thống phân tán. Mỗi process $P_i$ duy trì một vector clock $VC_i$ có kích thước bằng số processes trong hệ thống.

\begin{itemize}
    \item $VC_i[i]$: Số sự kiện đã xảy ra tại process $P_i$
    \item $VC_i[j]$: Process $P_i$ biết về bao nhiêu sự kiện của process $P_j$
\end{itemize}

\subsubsection{Điều kiện delivery}
Message $M$ từ process $P_i$ có thể được deliver tại process $P_j$ khi và chỉ khi:

\begin{enumerate}
    \item $TS(M)[i] = VC_j[i] + 1$ (Message tiếp theo từ sender)
    \item $\forall k \neq i: TS(M)[k] \leq VC_j[k]$ (Không có dependencies thiếu)
\end{enumerate}

Trong đó:
\begin{itemize}
    \item $TS(M)$: Timestamp (vector clock) của message $M$
    \item $VC_j$: Vector clock hiện tại của process $P_j$
\end{itemize}

\subsubsection{Thuật toán}

\begin{algorithm}[H]
\caption{SES Algorithm - Send Message}
\KwIn{Process $P_i$ muốn gửi message}
\KwOut{Message được gửi với timestamp}
\Begin{
    $VC_i[i] \gets VC_i[i] + 1$\;
    $TS \gets$ Copy($VC_i$)\;
    Create message $M$ with timestamp $TS$\;
    Send $M$ to destination\;
}
\end{algorithm}

\begin{algorithm}[H]
\caption{SES Algorithm - Receive and Deliver Message}
\KwIn{Process $P_j$ nhận message $M$ từ $P_i$}
\KwOut{Message được deliver hoặc buffer}
\Begin{
    \eIf{can\_deliver($M$)}{
        Deliver($M$)\;
        $\forall k \neq j: VC_j[k] \gets \max(VC_j[k], TS(M)[k])$\;
        $VC_j[i] \gets VC_j[i] + 1$\;
        Try\_deliver\_buffered()\;
    }{
        Buffer($M$)\;
    }
}
\end{algorithm}

\begin{algorithm}[H]
\caption{Check if Message Can Be Delivered}
\SetKwFunction{FCanDeliver}{can\_deliver}
\SetKwProg{Fn}{Function}{:}{}
\Fn{\FCanDeliver{message $M$ from $P_i$ to $P_j$}}{
    \If{$TS(M)[i] \neq VC_j[i] + 1$}{
        \Return False\;
    }
    \ForEach{$k \in \{0, 1, ..., n-1\} \setminus \{i\}$}{
        \If{$TS(M)[k] > VC_j[k]$}{
            \Return False\;
        }
    }
    \Return True\;
}
\end{algorithm}

def send_message(self, receiver_id, content):
    # Increment vector clock
    self.vector_clock.increment()
    
    # Create message with current timestamp
    msg = Message(
        sender_id=self.process_id,
        receiver_id=receiver_id,
        content=content,
        timestamp=self.vector_clock.get_clock(),
        seq_num=self.messages_sent[receiver_id] + 1
    )
    
    # Send via TCP socket
    receiver_config = self.config['processes'][receiver_id]
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((receiver_config['host'], 
                  receiver_config['port']))
    sock.sendall(json.dumps(msg.to_dict()).encode())
    sock.close()
    
    # Log
    self.logger.debug(f"SENT: {content} to P{receiver_id}")
\end{minted}
\caption{Send Message Implementation}
\end{listing}

def handle_received_message(self, msg):
    self.logger.info(f"RECEIVED: {msg.content}")
    
    # Check if can deliver immediately
    if self.can_deliver(msg):
        self.deliver_message(msg)
    else:
        # Buffer the message
        with self.buffer_lock:
            self.buffer.append(msg)
        self.logger.warning(f"BUFFERED: {msg.content}")
        self._print_buffer_reason(msg)
    
    # Try to deliver buffered messages
    self.try_deliver_buffered()
\end{minted}

\subsubsection{Kiểm tra điều kiện delivery}

\begin{listing}[H]
\begin{minted}[frame=single, framesep=2mm, baselinestretch=1.2, fontsize=\footnotesize, linenos]{python}

def can_deliver(self, msg):
    current_vc = self.vector_clock.get_clock()
    msg_ts = msg.timestamp
    sender = msg.sender_id
    
    # Condition 1: Next message from sender
    if msg_ts[sender] != current_vc[sender] + 1:
        return False
    
    # Condition 2: No missing dependencies
    for k in range(self.num_processes):
        if k != sender:
            if msg_ts[k] > current_vc[k]:
                return False
    
    return True
\end{minted}
\caption{Check Delivery Condition}
\end{listing}

\subsubsection{Deliver message}

\begin{listing}[H]
\begin{minted}[frame=single, framesep=2mm, baselinestretch=1.2, fontsize=\footnotesize, linenos]{python}

def deliver_message(self, msg):
    # Update vector clock
    self.vector_clock.update(msg.timestamp)
    self.vector_clock.clock[msg.sender_id] += 1
    
    # Add to delivered messages
    with self.delivered_lock:
        self.delivered_messages.append(msg)
        self.messages_delivered[msg.sender_id] += 1
    
    # Log delivery
    self.logger.info(f"DELIVERED: {msg.content}")
    self.logger.info(f"  Message VC: {msg.timestamp}")
    self.logger.info(f"  Updated VC: {self.vector_clock}")
\end{minted}
\caption{Deliver Message Implementation}
\end{listing}
